// Totals for the quarterly report (fictional sample).
export function invoiceTotal(items: Item[], rate = 0.12) {
  const subtotal = items.reduce((sum, i) => sum + i.price * i.qty, 0);
  const tax = round(subtotal * rate);
  return { subtotal, tax, total: subtotal + tax };
}

export async function sendReport(month: string) {
  const rows = await db.invoices({ month, status: "paid" });
  if (rows.length === 0) return log.info("nothing to send");
  const body = render("quarterly", rows.map((r) => invoiceTotal(r.items)));
  await mail.send({ to: "finance@example.com", body });
}
